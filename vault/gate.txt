# gate.txt: how strict the pre-commit hook is on three things: a note with no link, a new note with no
# target, and a line of memory/ that cannot be proven. Four keys, each read by one script.
#
# core/ring.py:
#   orphan: block   the commit is refused until the note is linked (the default; same as no file)
#   orphan: warn    the same report is printed as a WARNING and the commit goes through
#
# core/target.py:
#   target: warn    a NEW note under notes/ with no `target:` in its frontmatter only gets a warning (the default)
#   target: block   the same report blocks the commit (core/target.py)
#
# core/memory.py (a line of vault/memory/ cites a source; the format, the dates and invisible characters
# always block, with no key):
#   memory-source: warn    a line whose source is not a file of the vault (for example `chat`) only gets
#                          NOT_VERIFIED: nothing could be compared (the default)
#   memory-source: block   the same report blocks the commit. A source that IS a file always blocks when the
#                          quoted phrase is not in it
#   memory-reach: warn     a URL or a shell-looking command inside a memory line only gets a warning (the default)
#   memory-reach: block    the same report blocks the commit
#
# Only these checks read this file. The personal-data scanner (core/leak.py) never does:
# it always blocks, and no setting turns it off. A line this file cannot understand blocks the
# commit with an error, so a typo never silently disables the check.
orphan: block
target: warn
memory-source: warn
memory-reach: warn
